# SCOPE: this file contains ONLY what the live audio pipeline needs to run

# cleanly on the Pi -- if it doesn't, the demo cannot run. Optional extras
# for offline/dataset work live in requirements-optional.txt precisely so a
# resolver failure there can never block the demo path.
# See progress.md's 2026-08-24 dependency entry for why this split exists.

matplotlib>=3.5.0

# deepfilternet 0.5.6 hard-requires numpy<2.0. It is NOT listed in this file
# (assumed pre-installed, see above), so pip's resolver cannot see that
# constraint and will happily install numpy 2.x to satisfy some other
# package, silently breaking deepfilternet -- this happened on the Pi
# 2026-08-24 (numpy 1.26.4 -> 2.5.2, with pip printing a dependency-conflict
# warning but installing anyway).
#
# EXACT pin, not a range: an open range (numpy>=1.22.0,<2.0) let pip's
# backtracking resolver wander into older numpy patch versions while jointly
# resolving with other packages, even though 1.26.4 already satisfied
# everything and was found first -- it ended up trying to BUILD numpy 1.25.2
# from source (no aarch64 wheel for that patch) and failing on Python 3.13
# (pkg_resources/setuptools: `pkgutil.ImpImporter` was removed). An exact pin
# removes numpy as a backtracking axis entirely. 1.26.4 has a prebuilt
# cp313/aarch64 wheel and is the version the Pi already runs.
numpy==1.26.4

# Phase 1: reference-NLMS adaptive filter uses numba for the JIT hot loop.
# Pinned to the exact wheel verified on Pi (cp313/aarch64 exists for 0.67.0).
# This promotes numba from requirements-optional.txt (where it lived as an
# optional feature) to a core requirement now that dual-mic is a headline
# demo feature.  Must coexist with numpy==1.26.4 (confirmed in
# requirements-optional.txt comments).
# T0 gate: verify `pip install numba==0.67.0` succeeds on the Pi before
# running any dual-mic demo.  If it fails, downgrade this to optional and
# implement a pure-numpy fallback.
numba==0.67.0
