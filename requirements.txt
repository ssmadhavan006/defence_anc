# requirements.txt
# PS26052 Phase 5 — Raspberry Pi 5 Live Pipeline requirements.
# torch and deepfilternet are assumed to be pre-installed on the Pi.

sounddevice>=0.4.6
soundfile>=0.12.1
# deepfilternet 0.5.6 hard-requires numpy<2.0. It is NOT listed in this file
# (assumed pre-installed, see above), so pip's resolver cannot see that
# constraint on its own and will happily pick numpy 2.x to satisfy onnx/
# onnxruntime below, silently breaking deepfilternet -- this actually
# happened on the Pi 2026-08-24 (`pip install -r requirements.txt` upgraded
# numpy 1.26.4 -> 2.5.2, pip printed a dependency-conflict warning but
# installed it anyway).
#
# EXACT pin, not a range: an open range (numpy>=1.22.0,<2.0) let pip's
# backtracking resolver wander into older numpy patch versions while trying
# to jointly satisfy scipy/numba/onnx's ranges, even though numpy 1.26.4
# already satisfied everything and was found first -- it ended up trying to
# BUILD numpy 1.25.2 from source (no aarch64 wheel for that patch version)
# and failed outright on Python 3.13 due to an unrelated pkg_resources/
# setuptools incompatibility (`pkgutil.ImpImporter` was removed). An exact
# pin removes numpy as a backtracking axis entirely -- confirmed 1.26.4 has
# a prebuilt aarch64/cp313 wheel on piwheels (this exact version was already
# cached on the Pi from before this file existed).
numpy==1.26.4
pyyaml>=6.0
psutil>=5.9.0
matplotlib>=3.5.0

# scipy/numba: EXACT pins, same reasoning as numpy above. Open ranges
# (scipy>=1.10.0, numba>=0.58.0) made pip's resolver backtrack through many
# candidate versions of BOTH simultaneously (confirmed on the Pi 2026-08-24
# it fetched metadata for numba 0.67.0 down through 0.60.0), eventually
# reaching numba 0.60.0, which has no prebuilt wheel and fails to even
# build on Python 3.13 (`RuntimeError: Cannot install on Python version
# 3.13.5; only versions >=3.9,<3.13 are supported`) -- despite numba 0.67.0
# already having a confirmed-available cp313/aarch64 wheel. Pinned to the
# exact versions pip's own output showed succeeding at the metadata-fetch
# step (i.e. confirmed to have a real wheel for this platform).
scipy==1.18.1
numba==0.67.0

# P1-3: optional ONNX Runtime inference backend (live/inference_engine.py
# backend="onnx"). Only needed if you run models/deepfilternet/export_onnx.py
# and enable pipeline.inference_backend: onnx in audio_config.yaml — the
# default PyTorch backend needs neither of these. Confirmed compatible with
# numpy<2.0 (dev-machine `uv` resolution keeps numpy at 1.26.4 with these
# installed) -- if pip ever wants to bump numpy past 2.0 to satisfy these,
# that means a newer onnx/onnxruntime release changed its floor; re-verify
# against deepfilternet's numpy<2.0 constraint before allowing it.
onnx>=1.22.0
onnxruntime>=1.29.0
# Needed even though models/deepfilternet/export_onnx.py never uses torch's
# dynamo exporter (dynamo=True) -- on newer torch releases (confirmed: the
# Pi's), torch.onnx.export() unconditionally imports pieces of the dynamo
# exporter machinery internally at import time, regardless of which export
# path actually runs. Missing this produced `ModuleNotFoundError: No module
# named 'onnxscript'` from inside torch.onnx.export() itself on the Pi,
# despite working fine without it on the dev machine's older torch version.
onnxscript>=0.7.1
